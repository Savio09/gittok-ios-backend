  var in Input
  if name=="charmbracelet__glow.json" || name=="rack__rack.json" {
   var c struct { Repo githubclient.Repo `json:"metadata"`; CapturedAt time.Time `json:"captured_at"`; Revision crawler.Revision `json:"revision"`; Tree githubclient.FileTree `json:"tree"`; Fetched githubclient.FetchedContents `json:"consumed"` }
   raw,err:=os.ReadFile(filepath.Join("..","analysis","testdata","g17",name));if err!=nil {t.Fatal(err)}

   in=Input{Repo:c.Repo,CapturedAt:c.CapturedAt,Inspection:crawler.NewInspection(c.Repo.Language,c.Revision,c.Tree,c.Fetched)}
  } else { in=corpusInput(t,name) }
  p:=Build(in)
  for _,c:=range claimsIn(p){ if err:=ValidateClaim(in,c);err!=nil {t.Fatal(err)} }
  raw,err:=json.MarshalIndent(p,"","  ");if err!=nil {t.Fatal(err)}
  if err=os.WriteFile(filepath.Join("/Users/fdeclan/Public/gittok-ios-backend/docs/reviews/2026-11-04-g17-six-issue-implementation/previews",name),raw,0600);err!=nil {t.Fatal(err)}
  t.Logf("%s: kind=%s claims=%d deps=%v",name,p.Kind,len(claimsIn(p)),depLabels(p))
 }
}
