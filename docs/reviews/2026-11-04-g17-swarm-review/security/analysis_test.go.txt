type securityTransport func(*http.Request) (*http.Response,error)
func (f securityTransport) RoundTrip(r *http.Request)(*http.Response,error){return f(r)}

func TestSecurityOverlayPolicyChangesAcrossPipeline(t *testing.T) {
 for _,stage:=range []string{"metadata","ref","tree","content1","content2","revalidation"} {
  for _,attack:=range []string{"scope expansion","missing scope evidence","revocation"} {
   t.Run(stage+"/"+attack,func(t *testing.T){
    metas,files:=0,0
    token:="ghp_"+strings.Repeat("0",36)
    transport:=securityTransport(func(r *http.Request)(*http.Response,error){
     if r.Header.Get("Authorization")!="Bearer "+token { t.Fatal("credential omitted") }
     current,body:="",`{}`
     switch {
     case r.URL.Path=="/rate_limit":
     case r.URL.Path=="/repos/o/r":
      metas++; current="metadata"; if metas>1 {current="revalidation"}
      body=`{"id":7,"owner":{"login":"o"},"name":"r","full_name":"o/r","private":false,"default_branch":"main"}`
     case strings.Contains(r.URL.Path,"/commits/"):
      current="ref"; body=`{"sha":"0123456789abcdef0123456789abcdef01234567"}`
     case strings.Contains(r.URL.Path,"/git/trees/"):
      current="tree"; body=`{"tree":[{"path":"package.json","type":"blob","mode":"100644"},{"path":"README.md","type":"blob","mode":"100644"}]}`
     case strings.Contains(r.URL.Path,"/contents/"):
      files++; current=fmt.Sprintf("content%d",files)
      content:=`{"name":"o","exports":"./index.js"}`
      if strings.HasSuffix(r.URL.Path,"README.md"){content="# SENTINEL\n\nPublic sentinel README."}
      body=fmt.Sprintf(`{"type":"file","encoding":"base64","content":%q}`,base64.StdEncoding.EncodeToString([]byte(content)))
     default:t.Fatalf("unexpected request %s",r.URL.Path)
     }
     h:=http.Header{"X-Oauth-Scopes":[]string{""}}
     status:=200
     if current==stage {switch attack {case "scope expansion":h["X-Oauth-Scopes"]=[]string{"","repo"};case "missing scope evidence":delete(h,"X-Oauth-Scopes");case "revocation":status=401;body=`{"message":"private provider diagnostics"}`}}
     return &http.Response{StatusCode:status,Header:h,Body:io.NopCloser(strings.NewReader(body)),Request:r},nil
    })
    client,err:=githubclient.NewPublic(token,githubclient.WithHTTPClient(&http.Client{Transport:transport}),githubclient.WithMaxRetries(0));if err!=nil{t.Fatal(err)}
    if err=client.VerifyCredential(context.Background());err!=nil{t.Fatal(err)}
    svc,err:=NewPublicServiceFromClient(client);if err!=nil{t.Fatal(err)}
    cache,err:=OpenPreviewCache(t.TempDir(),DefaultCacheLimits(),time.Now);if err!=nil{t.Fatal(err)};svc.UseCache(cache)
    got,err:=svc.Preview(context.Background(),"o","r")
    if !errors.Is(err,githubclient.ErrCredentialPolicy){t.Fatalf("not terminal: %+v %v",got,err)}
    if got.ID!=""||cache.Bytes()!=0 {t.Fatalf("policy failure returned/stored data: %+v, bytes %d",got,cache.Bytes())}
   })
  }
 }
}

func TestSecurityOverlayWarmCacheRejectsExpandedScopes(t *testing.T){
 var reject bool
 calls:=0
 token:="ghp_"+strings.Repeat("0",36)
 transport:=securityTransport(func(r *http.Request)(*http.Response,error){
  calls++;h:=http.Header{"X-Oauth-Scopes":[]string{""}};if reject {h["X-Oauth-Scopes"]=[]string{"repo"}}
  body:=`{"id":7,"owner":{"login":"o"},"name":"r","full_name":"o/r","private":false,"default_branch":"main"}`
  if strings.Contains(r.URL.Path,"/commits/"){body=`{"sha":"0123456789abcdef0123456789abcdef01234567"}`}
  if strings.Contains(r.URL.Path,"/git/trees/"){body=`{"tree":[]}`}
  return &http.Response{StatusCode:200,Header:h,Body:io.NopCloser(strings.NewReader(body)),Request:r},nil
 })
 c,err:=githubclient.NewPublic(token,githubclient.WithHTTPClient(&http.Client{Transport:transport}),githubclient.WithMaxRetries(0));if err!=nil{t.Fatal(err)}
 s,err:=NewPublicServiceFromClient(c);if err!=nil{t.Fatal(err)}
 cache,err:=OpenPreviewCache(t.TempDir(),DefaultCacheLimits(),time.Now);if err!=nil{t.Fatal(err)};s.UseCache(cache)
 if _,err=s.Preview(context.Background(),"o","r");err!=nil{t.Fatal(err)}
 if cache.Bytes()==0{t.Fatal("setup did not warm cache")}
 oldCalls:=calls;reject=true
 p,err:=s.Preview(context.Background(),"o","r")
 if !errors.Is(err,githubclient.ErrCredentialPolicy)||p.ID!=""||calls-oldCalls!=1{t.Fatalf("warm policy gate failed: %+v %v calls=%d",p,err,calls-oldCalls)}
}
