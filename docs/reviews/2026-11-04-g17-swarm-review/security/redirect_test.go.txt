package githubclient

import (
 "context"
 "errors"
 "io"
 "net/http"
 "strings"
 "testing"
)

type reviewTransport func(*http.Request)(*http.Response,error)
func (f reviewTransport) RoundTrip(r *http.Request)(*http.Response,error){return f(r)}

func TestSecurityOverlayOriginChangesAreTerminal(t *testing.T){
 for _,location:=range []string{"http://api.github.com/repos/o/r","https://api.github.com:444/repos/o/r","https://api.github.com.evil.example/repos/o/r","https://github.com/repos/o/r","https://evil.example/repos/o/r"}{
  t.Run(location,func(t *testing.T){
   requests:=0
   rt:=reviewTransport(func(r *http.Request)(*http.Response,error){
    requests++; if requests>1{t.Fatalf("off-origin request was sent to %s",r.URL)}
    return &http.Response{StatusCode:302,Header:http.Header{"Location":[]string{location}},Body:io.NopCloser(strings.NewReader(`private redirect body`)),Request:r},nil
   })
   c,err:=NewPublic(fakeClassicPAT(),WithHTTPClient(&http.Client{Transport:rt}));if err!=nil{t.Fatal(err)}
   got,err:=c.GetRepo(context.Background(),"o","r")
   if !errors.Is(err,ErrCredentialPolicy)||got.FullName!=""||requests!=1{t.Fatalf("redirect not terminal: %+v %v requests=%d",got,err,requests)}
   if strings.Contains(err.Error(),"private redirect body"){t.Fatal("provider body exposed")}
  })
 }
}
